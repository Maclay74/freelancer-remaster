# A pragma wrapper is understood by both rc.exe and CMake's clang-cl dependency
# scanner, which otherwise forwards rc.exe's /c65001 switch to the C++ compiler.
get_property(fl_subdirs DIRECTORY PROPERTY SUBDIRECTORIES)
function(fl_utf8_resources directory)
  get_property(targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
  foreach(target IN LISTS targets)
    get_target_property(sources ${target} SOURCES)
    get_target_property(source_dir ${target} SOURCE_DIR)
    get_target_property(binary_dir ${target} BINARY_DIR)
    set(updated)
    foreach(source IN LISTS sources)
      if(source MATCHES "\\.rc$")
        get_filename_component(absolute "${source}" ABSOLUTE BASE_DIR "${source_dir}")
        get_filename_component(stem "${source}" NAME_WE)
        set(wrapper "${binary_dir}/${target}-${stem}-utf8.rc")
        file(CONFIGURE OUTPUT "${wrapper}" CONTENT "#pragma code_page(65001)\n#include \"${absolute}\"\n")
        list(APPEND updated "${wrapper}")
      else()
        list(APPEND updated "${source}")
      endif()
    endforeach()
    if(updated)
      set_property(TARGET ${target} PROPERTY SOURCES "${updated}")
    endif()
  endforeach()
endfunction()
foreach(directory IN LISTS fl_subdirs)
  fl_utf8_resources("${directory}")
endforeach()
